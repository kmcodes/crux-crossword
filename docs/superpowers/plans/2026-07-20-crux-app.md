# Crux App Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Crux, an offline Flutter crossword player that reads the pipeline's `puzzles.sqlite` pack, lets you leave any puzzle mid-solve and resume it later, and shows every in-progress puzzle with a completion percentage.

**Architecture:** Read-only puzzle pack bundled as a Flutter asset; a separate writable progress database so regenerating the pack never destroys in-progress solves. Grid rendered with `CustomPaint` for crisp zooming on 21x21. Provider for state, GoRouter for navigation.

**Tech Stack:** Flutter 3.41.6 / Dart 3.11.4 (verified installed), `sqlite3` + `sqlite3_flutter_libs` for storage, `provider`, `go_router`, `path_provider`.

**Depends on:** `docs/superpowers/plans/2026-07-20-pipeline.md` (produces the pack). A pack must exist at `assets/puzzles.sqlite` before Task 3.

**Spec:** `docs/superpowers/specs/2026-07-20-crossworder-design.md`

## Global Constraints

- Flutter 3.41.6, Dart SDK `>=3.11.0 <4.0.0`.
- App name and title: **Crux**.
- Fully offline. No network calls anywhere in the app.
- The pack database is opened **read-only** and never written to.
- Progress lives in a **separate database file** (`progress.sqlite`) in the app documents directory.
- Autosave on every letter entry — leaving a puzzle must never lose state.
- Completion % = filled non-black cells / total non-black cells.
- Use `const` constructors wherever possible; business logic in providers, not widgets.
- Check `mounted` after every async gap before touching `BuildContext`.
- No `print` — use `debugPrint` or a logger.

```dart
class Puzzle {
  const Puzzle({
    required this.id,
    required this.size,
    required this.difficulty,
    required this.grid,
    required this.solution,
  });

  final int id;
  final int size;
  final int difficulty;

  /// Black-square pattern: '#' for black, '.' for fillable.
  final String grid;

  /// Correct letters, same indexing as [grid].
  final String solution;

  bool isBlack(int index) => grid[index] == '#';

  int get fillableCount =>
      grid.split('').where((ch) => ch != '#').length;
}

class Clue {
  const Clue({
    required this.direction,
    required this.number,
    required this.row,
    required this.col,
    required this.length,
    required this.text,
    required this.answer,
  });

  /// 'A' for across, 'D' for down.
  final String direction;
  final int number;
  final int row;
  final int col;
  final int length;
  final String text;
  final String answer;

  bool get isAcross => direction == 'A';

  /// Flat cell indices this clue covers, for a grid of [size] columns.
  List<int> cells(int size) => [
        for (var i = 0; i < length; i++)
          isAcross ? row * size + col + i : (row + i) * size + col,
      ];
}
```

`app/lib/data/pack_database.dart`:
```dart
import 'package:sqlite3/sqlite3.dart';

import '../models/puzzle.dart';

/// Read-only access to the generated puzzle pack.
///
/// The pack is never written to; all mutable state lives in
/// [ProgressDatabase] so regenerating the pack cannot destroy progress.
class PackDatabase {
  PackDatabase._(this._db);

  final Database _db;

  factory PackDatabase.open(String path) =>
      PackDatabase._(sqlite3.open(path, mode: OpenMode.readOnly));

  List<Puzzle> allPuzzles() {
    final rows = _db.select(
      'SELECT id, size, difficulty, grid, solution FROM puzzles '
      'ORDER BY difficulty, id',
    );
    return rows.map(_puzzleFrom).toList();
  }

  Puzzle puzzleById(int id) {
    final rows = _db.select(
      'SELECT id, size, difficulty, grid, solution FROM puzzles WHERE id = ?',
      [id],
    );
    if (rows.isEmpty) {
      throw StateError('No puzzle with id $id in pack');
    }
    return _puzzleFrom(rows.first);
  }

  List<Clue> cluesFor(int puzzleId) {
    final rows = _db.select(
      'SELECT direction, number, row, col, length, clue, answer '
      'FROM puzzle_clues WHERE puzzle_id = ? ORDER BY number, direction',
      [puzzleId],
    );
    return rows
        .map((r) => Clue(
              direction: r['direction'] as String,
              number: r['number'] as int,
              row: r['row'] as int,
              col: r['col'] as int,
              length: r['length'] as int,
              text: r['clue'] as String,
              answer: r['answer'] as String,
            ))
        .toList();
  }

  Puzzle _puzzleFrom(Row r) => Puzzle(
        id: r['id'] as int,
        size: r['size'] as int,
        difficulty: r['difficulty'] as int,
        grid: r['grid'] as String,
        solution: r['solution'] as String,
      );

  // sqlite3 deprecated Database.dispose() in favour of close().
  void dispose() => _db.close();
}
```

```dart
import 'puzzle.dart';

/// Cell and clue navigation for one puzzle. Pure Dart on purpose: this is the
/// logic most prone to off-by-one bugs, so it is tested without any widgets.
class GridGeometry {
  GridGeometry(this.puzzle, this.clues) {
    for (final clue in clues) {
      for (final cell in clue.cells(puzzle.size)) {
        (clue.isAcross ? _acrossAt : _downAt)[cell] = clue;
      }
    }
    _across = clues.where((c) => c.isAcross).toList()
      ..sort((a, b) => a.number.compareTo(b.number));
    _down = clues.where((c) => !c.isAcross).toList()
      ..sort((a, b) => a.number.compareTo(b.number));
  }

  final Puzzle puzzle;
  final List<Clue> clues;

  final Map<int, Clue> _acrossAt = {};
  final Map<int, Clue> _downAt = {};
  late final List<Clue> _across;
  late final List<Clue> _down;

  Clue? clueAt(int cellIndex, {required bool across}) =>
      across ? _acrossAt[cellIndex] : _downAt[cellIndex];

  int firstCellOf(Clue clue) => clue.row * puzzle.size + clue.col;

  int? nextCell(int from, {required bool across}) {
    final clue = clueAt(from, across: across);
    if (clue == null) return null;
    final cells = clue.cells(puzzle.size);
    final position = cells.indexOf(from);
    if (position < 0 || position + 1 >= cells.length) return null;
    return cells[position + 1];
  }

  int? previousCell(int from, {required bool across}) {
    final clue = clueAt(from, across: across);
    if (clue == null) return null;
    final cells = clue.cells(puzzle.size);
    final position = cells.indexOf(from);
    if (position <= 0) return null;
    return cells[position - 1];
  }

  Clue? nextClue(Clue current) => _step(current, 1);

  Clue? previousClue(Clue current) => _step(current, -1);

  Clue? _step(Clue current, int delta) {
    final list = current.isAcross ? _across : _down;
    final index = list.indexWhere((c) => c.number == current.number);
    if (index < 0) return null;
    final target = index + delta;
    if (target < 0 || target >= list.length) return null;
    return list[target];
  }

  /// Fraction of fillable cells that hold a letter, 0.0 to 1.0.
  double completion(Map<int, String> entries) {
    final total = puzzle.fillableCount;
    if (total == 0) return 0;
    var filled = 0;
    for (final entry in entries.entries) {
      if (!puzzle.isBlack(entry.key) && entry.value.isNotEmpty) filled++;
    }
    return filled / total;
  }

  bool isCorrect(Map<int, String> entries) {
    for (var i = 0; i < puzzle.grid.length; i++) {
      if (puzzle.isBlack(i)) continue;
      if (entries[i] != puzzle.solution[i]) return false;
    }
    return true;
  }
}
```

`app/lib/data/progress_database.dart`:
```dart
import 'package:sqlite3/sqlite3.dart';

class ProgressSummary {
  const ProgressSummary({
    required this.puzzleId,
    required this.filledCells,
    required this.elapsedSeconds,
    required this.lastPlayed,
    required this.completed,
  });

  final int puzzleId;
  final int filledCells;
  final int elapsedSeconds;
  final DateTime lastPlayed;
  final bool completed;
}

/// Writable solve state, deliberately separate from the puzzle pack so that
/// replacing the pack cannot destroy in-progress solves.
class ProgressDatabase {
  ProgressDatabase._(this._db) {
    _db.execute('''
      CREATE TABLE IF NOT EXISTS progress (
        puzzle_id  INTEGER NOT NULL,
        cell_index INTEGER NOT NULL,
        letter     TEXT    NOT NULL,
        pencil     INTEGER NOT NULL DEFAULT 0,
        PRIMARY KEY (puzzle_id, cell_index)
      );
      CREATE TABLE IF NOT EXISTS puzzle_state (
        puzzle_id       INTEGER PRIMARY KEY,
        elapsed_seconds INTEGER NOT NULL DEFAULT 0,
        last_played     TEXT    NOT NULL,
        completed_at    TEXT
      );
    ''');
  }

  final Database _db;

  factory ProgressDatabase.openAt(String path) =>
      ProgressDatabase._(sqlite3.open(path));

  factory ProgressDatabase.inMemory() =>
      ProgressDatabase._(sqlite3.openInMemory());

  void setLetter(int puzzleId, int cellIndex, String letter,
      {bool pencil = false}) {
    _db.execute(
      'INSERT INTO progress (puzzle_id, cell_index, letter, pencil) '
      'VALUES (?, ?, ?, ?) '
      'ON CONFLICT(puzzle_id, cell_index) DO UPDATE SET '
      'letter = excluded.letter, pencil = excluded.pencil',
      [puzzleId, cellIndex, letter, pencil ? 1 : 0],
    );
    _touch(puzzleId);
  }

  void clearLetter(int puzzleId, int cellIndex) {
    _db.execute(
      'DELETE FROM progress WHERE puzzle_id = ? AND cell_index = ?',
      [puzzleId, cellIndex],
    );
    _touch(puzzleId);
  }

  Map<int, String> entriesFor(int puzzleId) {
    final rows = _db.select(
      'SELECT cell_index, letter FROM progress WHERE puzzle_id = ?',
      [puzzleId],
    );
    return {
      for (final r in rows) r['cell_index'] as int: r['letter'] as String,
    };
  }

  Set<int> pencilCellsFor(int puzzleId) {
    final rows = _db.select(
      'SELECT cell_index FROM progress WHERE puzzle_id = ? AND pencil = 1',
      [puzzleId],
    );
    return {for (final r in rows) r['cell_index'] as int};
  }

  void saveElapsed(int puzzleId, int seconds) {
    _touch(puzzleId);
    _db.execute(
      'UPDATE puzzle_state SET elapsed_seconds = ? WHERE puzzle_id = ?',
      [seconds, puzzleId],
    );
  }

  void markCompleted(int puzzleId) {
    _touch(puzzleId);
    _db.execute(
      'UPDATE puzzle_state SET completed_at = ? WHERE puzzle_id = ?',
      [DateTime.now().toIso8601String(), puzzleId],
    );
  }

  List<ProgressSummary> inProgress() {
    final rows = _db.select('''
      SELECT s.puzzle_id,
             s.elapsed_seconds,
             s.last_played,
             s.completed_at,
             (SELECT COUNT(*) FROM progress p
               WHERE p.puzzle_id = s.puzzle_id) AS filled
      FROM puzzle_state s
      ORDER BY s.last_played DESC
    ''');
    return rows
        .map((r) => ProgressSummary(
              puzzleId: r['puzzle_id'] as int,
              filledCells: r['filled'] as int,
              elapsedSeconds: r['elapsed_seconds'] as int,
              lastPlayed: DateTime.parse(r['last_played'] as String),
              completed: r['completed_at'] != null,
            ))
        .toList();
  }

  void _touch(int puzzleId) {
    _db.execute(
      'INSERT INTO puzzle_state (puzzle_id, elapsed_seconds, last_played) '
      'VALUES (?, 0, ?) '
      'ON CONFLICT(puzzle_id) DO UPDATE SET last_played = excluded.last_played',
      [puzzleId, DateTime.now().toIso8601String()],
    );
  }

  // sqlite3 deprecated Database.dispose() in favour of close().
  void dispose() => _db.close();
}
```

```dart
import '../data/pack_database.dart';
import '../data/progress_database.dart';
import '../models/grid_geometry.dart';
import '../models/puzzle.dart';

class SolveProvider extends ChangeNotifier {
  SolveProvider({required PackDatabase pack, required ProgressDatabase progress})
      : _pack = pack,
        _progress = progress;

  final PackDatabase _pack;
  final ProgressDatabase _progress;

  Puzzle? _puzzle;
  GridGeometry? _geometry;
  Map<int, String> _entries = {};
  Set<int> _pencilCells = {};
  int _selectedCell = 0;
  bool _across = true;
  bool _pencilMode = false;

  Puzzle? get puzzle => _puzzle;
  GridGeometry? get geometry => _geometry;
  Map<int, String> get entries => Map.unmodifiable(_entries);
  Set<int> get pencilCells => Set.unmodifiable(_pencilCells);
  int get selectedCell => _selectedCell;
  bool get across => _across;
  bool get pencilMode => _pencilMode;

  Clue? get currentClue => _geometry?.clueAt(_selectedCell, across: _across);

  double get completion => _geometry?.completion(_entries) ?? 0;

  bool get solved => _geometry?.isCorrect(_entries) ?? false;

  void open(int puzzleId) {
    final puzzle = _pack.puzzleById(puzzleId);
    _puzzle = puzzle;
    _geometry = GridGeometry(puzzle, _pack.cluesFor(puzzleId));
    _entries = _progress.entriesFor(puzzleId);
    _pencilCells = _progress.pencilCellsFor(puzzleId);
    _across = true;
    _selectedCell = _firstFillableCell(puzzle);
    notifyListeners();
  }

  void selectCell(int cellIndex) {
    final puzzle = _puzzle;
    if (puzzle == null || puzzle.isBlack(cellIndex)) return;
    if (cellIndex == _selectedCell) {
      _across = !_across;
    } else {
      _selectedCell = cellIndex;
    }
    notifyListeners();
  }

  void toggleDirection() {
    _across = !_across;
    notifyListeners();
  }

  void setPencilMode(bool value) {
    _pencilMode = value;
    notifyListeners();
  }

  void typeLetter(String letter) {
    final puzzle = _puzzle;
    final geometry = _geometry;
    if (puzzle == null || geometry == null) return;
    if (puzzle.isBlack(_selectedCell)) return;

    final upper = letter.toUpperCase();
    _entries[_selectedCell] = upper;
    if (_pencilMode) {
      _pencilCells.add(_selectedCell);
    } else {
      _pencilCells.remove(_selectedCell);
    }

    // Autosave on every keystroke: leaving mid-solve must never lose state.
    _progress.setLetter(puzzle.id, _selectedCell, upper, pencil: _pencilMode);

    final next = geometry.nextCell(_selectedCell, across: _across);
    if (next != null) _selectedCell = next;

    if (geometry.isCorrect(_entries)) {
      _progress.markCompleted(puzzle.id);
    }
    notifyListeners();
  }

  void backspace() {
    final puzzle = _puzzle;
    final geometry = _geometry;
    if (puzzle == null || geometry == null) return;

    if (_entries.containsKey(_selectedCell)) {
      _entries.remove(_selectedCell);
      _pencilCells.remove(_selectedCell);
      _progress.clearLetter(puzzle.id, _selectedCell);
    } else {
      final previous = geometry.previousCell(_selectedCell, across: _across);
      if (previous != null) {
        _selectedCell = previous;
        _entries.remove(previous);
        _pencilCells.remove(previous);
        _progress.clearLetter(puzzle.id, previous);
      }
    }
    notifyListeners();
  }

  void nextClue() => _jumpClue(forward: true);

  void previousClue() => _jumpClue(forward: false);

  void saveElapsed(int seconds) {
    final puzzle = _puzzle;
    if (puzzle == null) return;
    _progress.saveElapsed(puzzle.id, seconds);
  }

  void _jumpClue({required bool forward}) {
    final geometry = _geometry;
    final current = currentClue;
    if (geometry == null || current == null) return;
    final target =
        forward ? geometry.nextClue(current) : geometry.previousClue(current);
    if (target == null) return;
    _selectedCell = geometry.firstCellOf(target);
    notifyListeners();
  }

  int _firstFillableCell(Puzzle puzzle) {
    for (var i = 0; i < puzzle.grid.length; i++) {
      if (!puzzle.isBlack(i)) return i;
    }
    return 0;
  }
}
```

```dart
import 'package:flutter/foundation.dart';

import '../data/pack_database.dart';
import '../data/progress_database.dart';
import '../models/puzzle.dart';

class PuzzleListing {
  const PuzzleListing({
    required this.puzzle,
    required this.completion,
    required this.elapsedSeconds,
    required this.lastPlayed,
    required this.completed,
  });

  final Puzzle puzzle;
  final double completion;
  final int elapsedSeconds;
  final DateTime? lastPlayed;
  final bool completed;
}

class LibraryProvider extends ChangeNotifier {
  LibraryProvider({required PackDatabase pack, required ProgressDatabase progress})
      : _pack = pack,
        _progress = progress;

  final PackDatabase _pack;
  final ProgressDatabase _progress;

  List<PuzzleListing> _inProgress = [];
  List<Puzzle> _library = [];
  int? _difficultyFilter;
  int? _sizeFilter;

  List<PuzzleListing> get inProgress => List.unmodifiable(_inProgress);
  List<Puzzle> get library => List.unmodifiable(_library);
  int? get difficultyFilter => _difficultyFilter;
  int? get sizeFilter => _sizeFilter;

  void load() {
    final all = _pack.allPuzzles();
    final byId = {for (final p in all) p.id: p};
    final summaries = _progress.inProgress();

    _inProgress = [
      for (final s in summaries)
        if (byId.containsKey(s.puzzleId) && !s.completed)
          PuzzleListing(
            puzzle: byId[s.puzzleId]!,
            completion: byId[s.puzzleId]!.fillableCount == 0
                ? 0
                : s.filledCells / byId[s.puzzleId]!.fillableCount,
            elapsedSeconds: s.elapsedSeconds,
            lastPlayed: s.lastPlayed,
            completed: s.completed,
          ),
    ];

    final startedIds = {for (final s in summaries) s.puzzleId};
    _library = all.where((p) {
      if (startedIds.contains(p.id)) return false;
      if (_difficultyFilter != null && p.difficulty != _difficultyFilter) {
        return false;
      }
      if (_sizeFilter != null && p.size != _sizeFilter) return false;
      return true;
    }).toList();

    notifyListeners();
  }

  void setDifficultyFilter(int? value) {
    _difficultyFilter = value;
    load();
  }

  void setSizeFilter(int? value) {
    _sizeFilter = value;
    load();
  }

  Puzzle? randomHard() {
    final hard = _library.where((p) => p.difficulty >= 4).toList();
    if (hard.isEmpty) return _library.isEmpty ? null : _library.first;
    hard.shuffle();
    return hard.first;
  }
}
```

```dart
class CrosswordGrid extends StatelessWidget {
  const CrosswordGrid({
    super.key,
    required this.puzzle,
    required this.geometry,
    required this.entries,
    required this.pencilCells,
    required this.selectedCell,
    required this.across,

  final Set<int> pencilCells;
  final int selectedCell;
  final bool across;
  final ValueChanged<int> onCellTapped;

    return LayoutBuilder(

        final side = constraints.biggest.shortestSide;
        return GestureDetector(
          onTapUp: (details) {

            final row = (details.localPosition.dy / cellSize).floor();
            if (row < 0 || col < 0 || row >= puzzle.size || col >= puzzle.size) {
              return;
            }
            onCellTapped(row * puzzle.size + col);

            maxScale: 4,

              child: CustomPaint(
                size: Size.square(side),
                painter: _GridPainter(
                  puzzle: puzzle,
                  geometry: geometry,
                  entries: entries,
                  pencilCells: pencilCells,
                  selectedCell: selectedCell,
                  highlighted: highlighted,
                  theme: Theme.of(context),
                ),
              ),
            ),
          ),
        );
      },
    );
  }
}

class _GridPainter extends CustomPainter {
  _GridPainter({
    required this.puzzle,
    required this.geometry,
    required this.entries,
    required this.pencilCells,
    required this.selectedCell,
    required this.highlighted,
    required this.theme,
  });

  final Puzzle puzzle;
  final GridGeometry geometry;
  final Map<int, String> entries;
  final Set<int> pencilCells;
  final int selectedCell;
  final Set<int> highlighted;
  final ThemeData theme;

  @override
  void paint(Canvas canvas, Size size) {
    final cellSize = size.width / puzzle.size;
    final line = Paint()
      ..color = theme.colorScheme.outline
      ..strokeWidth = 1
      ..style = PaintingStyle.stroke;

    final numbers = _startNumbers();

    for (var index = 0; index < puzzle.grid.length; index++) {
      final row = index ~/ puzzle.size;
      final col = index % puzzle.size;
      final rect = Rect.fromLTWH(
        col * cellSize,
        row * cellSize,
        cellSize,
        cellSize,
      );

      final fill = Paint()..style = PaintingStyle.fill;
      if (puzzle.isBlack(index)) {
        fill.color = theme.colorScheme.onSurface;
      } else if (index == selectedCell) {
        fill.color = theme.colorScheme.primaryContainer;
      } else if (highlighted.contains(index)) {
        fill.color = theme.colorScheme.secondaryContainer;
      } else {
        fill.color = theme.colorScheme.surface;
      }
      canvas.drawRect(rect, fill);
      canvas.drawRect(rect, line);

      if (puzzle.isBlack(index)) continue;

      final number = numbers[index];
      if (number != null) {
        _paintText(
          canvas,
          '$number',
          Offset(rect.left + cellSize * 0.06, rect.top + cellSize * 0.04),
          cellSize * 0.24,
          theme.colorScheme.onSurfaceVariant,
        );
      }

      final letter = entries[index];
      if (letter != null && letter.isNotEmpty) {
        final isPencil = pencilCells.contains(index);
        _paintText(
          canvas,
          letter,
          Offset(rect.left, rect.top + cellSize * 0.22),
          cellSize * 0.58,
          isPencil
              ? theme.colorScheme.onSurfaceVariant
              : theme.colorScheme.onSurface,
          width: cellSize,
          center: true,
        );
      }
    }
  }

  Map<int, int> _startNumbers() {
    final result = <int, int>{};
    for (final clue in geometry.clues) {
      result[geometry.firstCellOf(clue)] = clue.number;
    }
    return result;
  }

  void _paintText(
    Canvas canvas,
    String text,
    Offset offset,
    double fontSize,
    Color color, {
    double? width,
    bool center = false,
  }) {
    final painter = TextPainter(
      text: TextSpan(
        text: text,
        style: TextStyle(
          fontSize: fontSize,
          color: color,
          fontWeight: FontWeight.w500,
        ),
      ),
      textDirection: TextDirection.ltr,
      textAlign: center ? TextAlign.center : TextAlign.left,
    )..layout(minWidth: 0, maxWidth: width ?? double.infinity);
    painter.paint(canvas, offset);
  }

  @override
  bool shouldRepaint(_GridPainter old) =>
      old.selectedCell != selectedCell ||
      old.highlighted != highlighted ||
      !identical(old.entries, entries) ||
      old.entries.length != entries.length;
}
```

```dart
class _SolveScreenState extends State<SolveScreen> {
  Timer? _timer;
  int _elapsed = 0;

  /// Captured in didChangeDependencies, not read in dispose(). By the time
  /// dispose() runs the element is deactivated and an ancestor lookup throws
  /// "Looking up a deactivated widget's ancestor is unsafe" — which would fire
  /// every time the user leaves a puzzle, the single most common action here.
  SolveProvider? _solve;

  @override
  void initState() {
    super.initState();
    _timer = Timer.periodic(const Duration(seconds: 1), (_) {
      _elapsed++;
      if (_elapsed % 10 == 0) {
        _solve?.saveElapsed(_elapsed);
      }
    });
  }

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();
    _solve = context.read<SolveProvider>();
  }

  @override
  void dispose() {
    _timer?.cancel();
    // Persist the final time when leaving the puzzle.
    _solve?.saveElapsed(_elapsed);
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final solve = context.watch<SolveProvider>();
    final puzzle = solve.puzzle;
    final geometry = solve.geometry;

    if (puzzle == null || geometry == null) {
      return const Scaffold(body: Center(child: CircularProgressIndicator()));
    }

    final percent = (solve.completion * 100).round();

    return Scaffold(
      appBar: AppBar(
        title: Text('Crux  ·  $percent%'),
        actions: [
          IconButton(
            tooltip: solve.pencilMode ? 'Pencil on' : 'Pencil off',
            icon: Icon(solve.pencilMode ? Icons.edit : Icons.edit_outlined),
            onPressed: () => solve.setPencilMode(!solve.pencilMode),
          ),
        ],
      ),
      body: Column(
        children: [
          Expanded(
            child: Center(
              child: CrosswordGrid(
                puzzle: puzzle,
                geometry: geometry,
                entries: solve.entries,
                pencilCells: solve.pencilCells,
                selectedCell: solve.selectedCell,
                across: solve.across,
                onCellTapped: solve.selectCell,
              ),
            ),
          ),
          ClueBar(
            clue: solve.currentClue,
            onPrevious: solve.previousClue,
            onNext: solve.nextClue,
          ),
          CrosswordKeyboard(
            onLetter: solve.typeLetter,
            onBackspace: solve.backspace,
          ),
        ],
      ),
    );
  }
}
```

```dart
  final documents = await getApplicationDocumentsDirectory();

  final packFile = File(packPath);

          create: (_) => LibraryProvider(pack: pack, progress: progress),
        ),
        ChangeNotifierProvider(
          create: (_) => SolveProvider(pack: pack, progress: progress),
```

```dart
import 'package:crux/data/pack_database.dart';
import 'package:crux/data/progress_database.dart';
import 'package:crux/providers/library_provider.dart';
import 'package:crux/providers/solve_provider.dart';
import 'package:flutter_test/flutter_test.dart';

void main() {
  test('leave a puzzle, see it in progress, resume it', () {
    final pack = PackDatabase.open('assets/puzzles.sqlite');
    final progress = ProgressDatabase.inMemory();
    final puzzle = pack.allPuzzles().first;

    // Start solving, type 10 correct letters, then walk away.
    final solve = SolveProvider(pack: pack, progress: progress)..open(puzzle.id);
    var typed = 0;
    for (var i = 0; i < puzzle.grid.length && typed < 10; i++) {
      if (puzzle.isBlack(i)) continue;
      solve.selectCell(i);
      solve.typeLetter(puzzle.solution[i]);
      typed++;
    }
    solve.saveElapsed(90);

    // The home screen lists it with a real completion percentage.
    final library = LibraryProvider(pack: pack, progress: progress)..load();
    expect(library.inProgress.length, 1);
    final listing = library.inProgress.first;
    expect(listing.puzzle.id, puzzle.id);
    expect(listing.completion, closeTo(10 / puzzle.fillableCount, 0.001));
    expect(listing.elapsedSeconds, 90);

    // And it no longer shows as unplayed.
    expect(library.library.any((p) => p.id == puzzle.id), isFalse);

    // Resuming restores every letter.
    final resumed = SolveProvider(pack: pack, progress: progress)
      ..open(puzzle.id);
    expect(resumed.entries.length, 10);
    for (final entry in resumed.entries.entries) {
      expect(entry.value, puzzle.solution[entry.key]);
    }

    pack.dispose();
    progress.dispose();
  });
}
```

## Open risk

`SolveProvider` is created once at app start and reused for whichever puzzle is
open. If a second solve screen is ever pushed without `open()` being called, it
would show the previous puzzle. The router calls `open()` on every navigation to
`/solve/:id`, so this holds today, but if the app later gains a way to reach the
solve screen without routing, that assumption needs revisiting.
